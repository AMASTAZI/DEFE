<script>
/* ── SYSTÈME DE MESSAGERIE — LOCALSTORAGE ── */
 
function genId() {
  return 'msg_' + Date.now() + '_' + Math.random().toString(36).substr(2,6);
}
 
function getMsgs() {
  return JSON.parse(localStorage.getItem('fse_messages') || '[]');
}
 
function saveMsgs(msgs) {
  localStorage.setItem('fse_messages', JSON.stringify(msgs));
}
 
function envoyerContact() {
  const prenom  = (document.getElementById('c-prenom')?.value || '').trim();
  const nom     = (document.getElementById('c-nom')?.value || '').trim();
  const email   = (document.getElementById('c-email')?.value || '').trim();
  const tel     = (document.getElementById('c-tel')?.value || '').trim();
  const profil  = (document.getElementById('c-profil')?.value || '').trim();
  const sujet   = (document.getElementById('c-sujet')?.value || '').trim();
  const message = (document.getElementById('c-message')?.value || '').trim();
 
  const succes  = document.getElementById('contact-succes');
  const erreur  = document.getElementById('contact-erreur');
  const errTxt  = document.getElementById('contact-erreur-txt');
  const btn     = document.getElementById('btn-contact-envoyer');
 
  // Reset alertes
  succes.style.display  = 'none';
  erreur.style.display  = 'none';
 
  // Validation
  if (!prenom || !nom) {
    errTxt.textContent = 'Veuillez renseigner votre prénom et nom.';
    erreur.style.display = 'flex';
    return;
  }
  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    errTxt.textContent = 'Veuillez saisir une adresse e-mail valide.';
    erreur.style.display = 'flex';
    return;
  }
  if (!message) {
    errTxt.textContent = 'Veuillez écrire votre message.';
    erreur.style.display = 'flex';
    return;
  }
 
  // Sauvegarde
  const msgs = getMsgs();
  msgs.push({
    id:      genId(),
    nom:     prenom + ' ' + nom,
    email:   email,
    tel:     tel,
    profil:  profil,
    sujet:   sujet,
    message: message,
    date:    new Date().toISOString(),
    statut:  'nouveau'
  });
  saveMsgs(msgs);
 
  // Animation bouton
  btn.innerHTML = '<i class="fas fa-check" style="margin-right:8px;"></i>Message envoyé !';
  btn.style.background = 'linear-gradient(135deg,#0B5C2E,#157A3E)';
  btn.disabled = true;
 
  // Afficher succès
  succes.style.display = 'flex';
 
  // Reset formulaire après 3s
  setTimeout(() => {
    ['c-prenom','c-nom','c-email','c-tel','c-sujet','c-message'].forEach(id => {
      const el = document.getElementById(id);
      if (el) el.value = '';
    });
    const sel = document.getElementById('c-profil');
    if (sel) sel.selectedIndex = 0;
    btn.innerHTML = '<i class="fas fa-paper-plane" style="margin-right:8px;"></i>Envoyer le Message';
    btn.style.background = '';
    btn.disabled = false;
    succes.style.display = 'none';
  }, 4000);
}
</script>
